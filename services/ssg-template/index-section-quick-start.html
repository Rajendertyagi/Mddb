Quick Start #

bash — docker 
# Run MDDB server
docker run -d \
  -p 11023:11023 \
  -p 11024:11024 \
  -v mddb-data:/data \
  tradik/mddb:latest

# Test the API
curl http://localhost:11023/health
bash — docker compose 
# Clone repository
git clone https://github.com/tradik/mddb.git
cd mddb

# Start all services (MDDB + Panel + MCP)
make docker-up

# Services available:
# - MDDB HTTP: http://localhost:11023
# - MDDB gRPC: localhost:11024
# - Web Panel: http://localhost:3000
# - MCP Server: http://localhost:9000
bash — binary 
# Download binary
wget https://github.com/tradik/mddb/releases/download/v2.9.13/mddbd-v2.9.13-linux-amd64.tar.gz
tar -xzf mddbd-v2.9.13-linux-amd64.tar.gz

# Run server
./mddbd

# Server starts on http://localhost:11023
bash — build from source 
# Clone repository
git clone https://github.com/tradik/mddb.git
cd mddb

# Build
make build

# Run
./bin/mddbd
